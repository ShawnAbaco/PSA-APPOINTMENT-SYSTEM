-- Summary counts
SELECT 'Total Appointments' as Metric, COUNT(*) as Count FROM appointments
UNION ALL
SELECT 'Single Appointments', COUNT(*) FROM appointments WHERE type = 'single'
UNION ALL
SELECT 'Multiple Appointments', COUNT(*) FROM appointments WHERE type = 'multiple'
UNION ALL
SELECT 'Total Clients', COUNT(*) FROM appointment_clients
UNION ALL
SELECT 'Pending', COUNT(*) FROM appointments WHERE status = 'pending'
UNION ALL
SELECT 'Confirmed', COUNT(*) FROM appointments WHERE status = 'confirmed'
UNION ALL
SELECT 'Completed', COUNT(*) FROM appointments WHERE status = 'completed'
UNION ALL
SELECT 'Cancelled', COUNT(*) FROM appointments WHERE status = 'cancelled';

-- Summary of today's appointments by status
SELECT 
    status,
    COUNT(*) as count,
    SUM(CASE WHEN type = 'single' THEN 1 ELSE 0 END) as single_appointments,
    SUM(CASE WHEN type = 'multiple' THEN 1 ELSE 0 END) as multiple_appointments
FROM appointments
WHERE appointment_date = CURDATE()
GROUP BY status;
